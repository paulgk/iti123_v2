   ```python
   def extract_features_v3(pose_sequence, apply_selection=True):
       # Step 1: Phase segmentation
       phases, validation = segment_and_validate(pose_sequence)

       # Step 2: V2 base features (spatial + temporal)
       # Reuse existing extract_frame_features and extract_temporal_features
       base_features = extract_v2_base_features(pose_sequence)

       # Step 3: P0 features (kinetic chain + contact)
       kinetic_chain = extract_kinetic_chain_timing(pose_sequence, phases)
       contact_features = extract_contact_frame_features(
           pose_sequence, phases['contact']
       )
       intent_features = extract_intent_window_features(
           pose_sequence, phases['contact']
       )
       sis = calculate_smash_intent_score(intent_features)

       # Step 4: P1 features (angular velocity + phase-specific)
       angular_vel = extract_angular_velocity_features(pose_sequence)
       phase_specific = extract_phase_specific_features(pose_sequence, phases)
       racket_speed = estimate_racket_head_speed(pose_sequence, phases)

       # Step 5: Combine all features
       all_features = {
           **base_features,
           **kinetic_chain,
           **contact_features,
           **intent_features,
           'smash_intent_score': sis['smash_intent_score'],
           **angular_vel,
           **phase_specific,
           **racket_speed,
           # Metadata
           'phase_validation_passed': validation['valid'],
           'feature_version': 'v3'
       }

       # Step 6: Apply feature selection (if enabled)
       if apply_selection:
           selected = load_selected_features()
           all_features = {k: v for k, v in all_features.items()
                          if k in selected or k.startswith('_meta')}

       return all_features
   ```

1. **FeatureEngineering class:**
   ```python
   class FeatureEngineering:
       """Feature engineering with version compatibility"""

       SUPPORTED_VERSIONS = ['v2', 'v3']

       def __init__(self, version='v3'):
           if version not in self.SUPPORTED_VERSIONS:
               raise ValueError(f"Version {version} not supported. Use: {self.SUPPORTED_VERSIONS}")
           self.version = version

       def extract_features(self, pose_sequence, **kwargs):
           """Extract features based on version"""
           if self.version == 'v2':
               return self._extract_v2(pose_sequence)
           elif self.version == 'v3':
               return self._extract_v3(pose_sequence, **kwargs)

       def _extract_v2(self, pose_sequence):
           """V2 extraction (427 features) - unchanged"""
           from src.data_processing.feature_engineering_v2 import (
               extract_temporal_features, extract_statistical_summary
           )
           per_frame, names, temporal = extract_temporal_features(pose_sequence)
           summary = extract_statistical_summary(per_frame, names)
           summary.update(temporal)
           return summary

       def _extract_v3(self, pose_sequence, apply_selection=True):
           """V3 extraction (< 254 selected features)"""
           from src.data_processing.feature_engineering_v3 import extract_features_v3
           return extract_features_v3(pose_sequence, apply_selection)

       def get_expected_features(self):
           """Return expected feature count for version"""
           if self.version == 'v2':
               return 427  # Or actual count
           elif self.version == 'v3':
               # Load from selection manifest
               selected = self._load_selected_features()
               return len(selected)
   ```

2. **get_feature_version(features_dict):**
   Detect version from feature dict:
   ```python
   def get_feature_version(features_dict):
       if 'feature_version' in features_dict:
           return features_dict['feature_version']
       elif 'smash_intent_score' in features_dict:
           return 'v3'  # SIS is v3-only
       else:
           return 'v2'  # Default to v2
   ```

3. **validate_feature_set(features_dict, expected_version):**
   Validate features match expected version:
   ```python
   def validate_feature_set(features_dict, expected_version):
       actual_version = get_feature_version(features_dict)
       if actual_version != expected_version:
           raise ValueError(f"Feature version mismatch: expected {expected_version}, got {actual_version}")

       expected_count = get_expected_feature_count(expected_version)
       actual_count = len(features_dict)
       if abs(actual_count - expected_count) > 10:  # Allow some tolerance
           raise ValueError(f"Feature count mismatch: expected ~{expected_count}, got {actual_count}")

       return True
   ```

```python
import pytest
import numpy as np
import json
import os

class TestFeatureEngineeringV3:

    @pytest.fixture
    def manifest_status(self):
        """Check if manifest is populated"""
        manifest_path = 'data/processed/features_v3/selected_features.json'
        if not os.path.exists(manifest_path):
            return 'missing'
        with open(manifest_path) as f:
            manifest = json.load(f)
        if manifest.get('selected_features'):
            return 'populated'
        return 'placeholder'

    def test_v3_extraction_imports(self):
        """Verify all modules can be imported"""
        from src.data_processing.feature_engineering_v3 import extract_features_v3
        from src.data_processing.feature_versioning import FeatureEngineering

    def test_v3_extraction_returns_dict(self):
        """Verify extraction returns dict with expected keys"""
        pose_sequence = np.random.randn(50, 33, 3)
        from src.data_processing.feature_engineering_v3 import extract_features_v3
        # Test WITHOUT selection (doesn't need manifest)
        features = extract_features_v3(pose_sequence, apply_selection=False)
        assert isinstance(features, dict)
        assert len(features) > 0

    def test_v3_includes_p0_features(self):
        """Verify P0 features present"""
        pose_sequence = np.random.randn(50, 33, 3)
        from src.data_processing.feature_engineering_v3 import extract_features_v3
        features = extract_features_v3(pose_sequence, apply_selection=False)
        assert 'hip_to_wrist_total' in features or any('kinetic' in k for k in features.keys())
        assert any('contact' in k for k in features.keys())

    def test_v3_includes_p1_features(self):
        """Verify P1 features present"""
        pose_sequence = np.random.randn(50, 33, 3)
        from src.data_processing.feature_engineering_v3 import extract_features_v3
        features = extract_features_v3(pose_sequence, apply_selection=False)
        assert any('angular' in k or 'vel' in k for k in features.keys())

    def test_v3_with_selection_reduces_count(self, manifest_status):
        """Verify selection reduces feature count to < 254"""
        if manifest_status != 'populated':
            pytest.skip(f"Manifest {manifest_status} - run feature_selection_pipeline first")

        pose_sequence = np.random.randn(50, 33, 3)
        from src.data_processing.feature_engineering_v3 import extract_features_v3
        features = extract_features_v3(pose_sequence, apply_selection=True)
        assert len(features) < 254

    def test_version_gating_v2(self):
        """Verify v2 extraction unchanged"""
        pose_sequence = np.random.randn(50, 33, 3)
        from src.data_processing.feature_versioning import FeatureEngineering
        fe_v2 = FeatureEngineering('v2')
        features_v2 = fe_v2.extract_features(pose_sequence)
        assert 'smash_intent_score' not in features_v2

    def test_version_detection(self):
        """Verify version detection from features"""
        from src.data_processing.feature_versioning import get_feature_version
        v2_features = {'max_velocity': 100, 'elbow_angle_mean': 90}
        v3_features = {'max_velocity': 100, 'smash_intent_score': 0.7}
        assert get_feature_version(v2_features) == 'v2'
        assert get_feature_version(v3_features) == 'v3'

    def test_backward_compatibility(self):
        """Verify v2 features subset of v3 (without selection)"""
        pose_sequence = np.random.randn(50, 33, 3)
        from src.data_processing.feature_versioning import FeatureEngineering
        fe_v2 = FeatureEngineering('v2')
        fe_v3 = FeatureEngineering('v3')
        features_v2 = fe_v2.extract_features(pose_sequence)
        features_v3 = fe_v3.extract_features(pose_sequence, apply_selection=False)
        v2_keys = set(features_v2.keys())
        v3_keys = set(features_v3.keys())
        overlap = len(v2_keys & v3_keys)
        assert overlap > len(v2_keys) * 0.5
```
